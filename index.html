-- CF32 - Permitir leitura dos itens das coletas finalizadas no histórico -- Não apaga nem altera as coletas existentes. alter table public.coleta_os_itens enable row level security; -- Remove somente políticas com estes nomes para permitir reexecução segura. drop policy if exists "ERP leitura coleta_os_itens anon" on public.coleta_os_itens; drop policy if exists "ERP leitura coleta_os_itens authenticated" on public.coleta_os_itens; create policy "ERP leitura coleta_os_itens anon" on public.coleta_os_itens for select to anon using (true); create policy "ERP leitura coleta_os_itens authenticated" on public.coleta_os_itens for select to authenticated using (true); grant select on table public.coleta_os_itens to anon, authenticated; -- Conferência: mostra quantos itens existem fisicamente por coleta. select c.id as coleta_id, c.quantidade as quantidade_cabecalho, count(i.id) as itens_salvos from public.coletas_os c left join public.coleta_os_itens i on i.coleta_id = c.id group by c.id, c.quantidade order by c.id desc limit 50;
